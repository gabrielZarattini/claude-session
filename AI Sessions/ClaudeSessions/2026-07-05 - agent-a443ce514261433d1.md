> 657	function parseMarkdownAndJson(text: string) {
> 658	  const trimmed = text.trim();
> 659	  // Try to parse global JSON first
> 660	  if ((trimmed.startsWith("{") && trimmed.endsWith("}")) || (trimmed.startsWith("[") && trimmed.endsWith("]"))) {
> 661	    try {
> 662	      const parsed = JSON.parse(trimmed);
> 663	      return { type: "json" as const, data: parsed };
> 664	    } catch {
> 665	      // Not a valid JSON, fallback to markdown
> 666	    }
> 667	  }
> 668	
> 669	  // Robust block splitter that supports streaming codeblocks
> 670	  const blocks: Array<{ type: "code" | "markdown"; content: string; lang?: string }> = [];
> 671	  let i = 0;
> 672	  while (i < text.length) {
> 673	    const nextCodeBlockStart = text.indexOf("```", i);
> 674	    if (nextCodeBlockStart === -1) {
> 675	      const content = text.slice(i);
> 676	      if (content) {
> 677	        blocks.push({ type: "markdown", content });
> 678	      }
> 679	      break;
> 680	    }
> 681	
> 682	    if (nextCodeBlockStart > i) {
> 683	      blocks.push({ type: "markdown", content: text.slice(i, nextCodeBlockStart) });
> 684	    }
> 685	
> 686	    const startOfCode = nextCodeBlockStart + 3;
> 687	    const endOfCode = text.indexOf("```", startOfCode);
> 688	    const newlineIndex = text.indexOf("\n", startOfCode);
> 689	    let lang = "";
> 690	    let codeStart = startOfCode;
> 691	
> 692	    if (newlineIndex !== -1 && (endOfCode === -1 || newlineIndex < endOfCode)) {
> 693	      lang = text.slice(startOfCode, newlineIndex).trim();
> 694	      codeStart = newlineIndex + 1;
> 695	    }
> 696	
> 697	    if (endOfCode === -1) {
> 698	      // Unclosed code block (streaming)
> 699	      const content = text.slice(codeStart);
> 700	      blocks.push({ type: "code", content, lang });
> 701	      break;
> 702	    } else {
> 703	      const content = text.slice(codeStart, endOfCode);
> 704	      blocks.push({ type: "code", content, lang });
> 705	      i = endOfCode + 3;
> 706	    }
> 707	  }
> 708	
> 709	  return { type: "markdown" as const, blocks };
> 710	}
