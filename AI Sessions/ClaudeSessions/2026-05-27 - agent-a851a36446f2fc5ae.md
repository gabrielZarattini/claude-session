CREATE OR REPLACE FUNCTION public.async_orchestrate_step(
  p_supabase_url text,
  p_service_jwt_legacy text,  -- new param: eyJ...
  p_run_id uuid,
  p_step text
)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
  PERFORM net.http_post(
    url     := p_supabase_url || '/functions/v1/orchestrate-step',
    headers := jsonb_build_object(
      'Content-Type',  'application/json',
