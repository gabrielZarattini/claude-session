async function getDefaultModel(userId?: string) {
  if (userId) {
    const { data } = await supabase
      .from("user_api_keys")
      .select("pref_ai_model")
      .eq("user_id", userId)
      .maybeSingle();
    if (data?.pref_ai_model) {
      return data.pref_ai_model;
    }
  }
  // Fallback to env-based
  const geminiKey = import.meta.env.VITE_GEMINI_API_KEY;
  const orKey = import.meta.env.VITE_OPENROUTER_API_KEY;
  if (geminiKey) return "gemini-2.5-flash";
  if (orKey?.startsWith("sk-or-")) return "google/gemini-2.0-flash-001";
  return "gemini-2.5-flash";
}
