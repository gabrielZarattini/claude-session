> 72	function xhrPost(body: Blob, headers: Record<string, string>, onLoaded: (loaded: number) => void): Promise<UploadResp> {
> 73	  return new Promise((resolve, reject) => {

> 76	    for (const [k, v] of Object.entries(headers)) xhr.setRequestHeader(k, v);
> 77	    xhr.upload.onprogress = (e) => { if (e.lengthComputable) onLoaded(e.loaded); };
> 78	    xhr.onload = () => {
> 79	      if (xhr.status === 200) {
> 80	        try { resolve(JSON.parse(xhr.responseText)); } catch { reject(new Error('Resposta inválida do servidor.')); }

> 97	export async function uploadMasterToHost(
> 98	  file: File,
> 99	  onProgress: (pct: number) => void,
> 100	): Promise<{ filename: string; bytes: number }> {
> 101	  const { data: { session } } = await supabase.auth.getSession();
> 102	  const token = session?.access_token;
> 103	  if (!token) throw new Error('Sessão expirada — faça login novamente.');
> 104	  // Headers must be ASCII — encode spaces/accents in the filename (server decodeURIComponent + sanitize).
> 105	  // X-Total-Bytes deixa o servidor VERIFICAR a integridade do arquivo montado antes de aceitá-lo.
> 106	  // Sem isso, um chunk perdido/duplicado produz um arquivo com tamanho plausível e mídia corrompida
> 107	  // (witnessed 2026-07-20: master do EP02 decodificava até ~2min e virava lixo depois).
> 108	  const base: Record<string, string> = {
> 109	    Authorization: `Bearer ${token}`,
> 110	    'X-Filename': encodeURIComponent(file.name),
> 111	    'X-Total-Bytes': String(file.size),
> 112	  };
> 113	  const total = file.size;
> 114	  onProgress(0);
> 115	
> 116	  if (total <= CHUNK_BYTES) {
> 117	    const r = await xhrPost(file, base, (loaded) => onProgress(Math.round((loaded / total) * 100)));
> 118	    return { filename: r.filename ?? file.name, bytes: r.bytes ?? total };
> 119	  }
> 120	
> 121	  const uploadId = crypto.randomUUID ? crypto.randomUUID() : `${Date.now()}-${Math.round(Math.random() * 1e9)}`;
> 122	  const count = Math.ceil(total / CHUNK_BYTES);
> 123	  let uploaded = 0;
> 124	  let result: UploadResp = {};
> 125	  for (let i = 0; i < count; i++) {
> 126	    const blob = file.slice(i * CHUNK_BYTES, Math.min((i + 1) * CHUNK_BYTES, total));
> 127	    result = await xhrPost(
> 128	      blob,
> 129	      { ...base, 'X-Upload-Id': uploadId, 'X-Chunk-Index': String(i), 'X-Chunk-Count': String(count) },
> 130	      (loaded) => onProgress(Math.round(((uploaded + loaded) / total) * 100)),
> 131	    );
> 132	    uploaded += blob.size;
> 133	    onProgress(Math.round((uploaded / total) * 100));
> 134	  }
> 135	  return { filename: result.filename ?? file.name, bytes: result.bytes ?? total };
> 136	}
