> 126	async function computeEquity() {
> 127	    const settings = await settingsRepository.getDefaultSettings(); // já vem DESCRIPTOGRAFADO (incl. id)
> 128	    if (!settings || !settings.id) return { valued: false, error: 'sem settings default' };
> 129	    const exchange = require('./utils/exchange')(settings);
> 130	    const bal = await exchange.balance();
> 131	    if (!bal) return { valued: false, error: 'saldo indisponível', settings };
> 132	
> 133	    let equity = 0;
> 134	    let usdtFree = 0;
> 135	    const cold = [];
> 136	    const ignored = [];
> 137	    for (const [coin, b] of Object.entries(bal)) {
> 138	        const qty = (parseFloat(b.available) || 0) + (parseFloat(b.onOrder) || 0);
> 139	        if (qty <= 0) continue;
> 140	        if (coin === QUOTE) { usdtFree = parseFloat(b.available) || 0; equity += qty; continue; }
> 141	        if (STABLES.has(coin)) { equity += qty; continue; }        // outras stables ~ $1
> 142	        if (IGNORE_ASSETS.has(coin)) { ignored.push(coin); continue; } // isento explícito pelo operador
> 143	        const px = priceOf(coin);
> 144	        if (px == null) {
> 145	            // 'LD'+ativo = wrapper do Binance Simple Earn (LDLTC = LTC em Earn, LDUSDC = USDC em Earn...). Tem valor
> 146	            // REAL e resgatável → marca a mercado pelo ativo SUBJACENTE (strip 'LD'). Sem isto, um saldo que a Binance
> 147	            // AUTO-SUBSCREVE no Earn some do equity e dispara o cap por falso drawdown (incidente 2026-06-20: LTC→LDLTC,
> 148	            // equity $148→$109 → HALT falso). length>3 preserva spot real como LDO.
> 149	            if (coin.startsWith('LD') && coin.length > 3) {
> 150	                const under = coin.slice(2);                            // LDLTC -> LTC, LDUSDC -> USDC
> 151	                if (STABLES.has(under)) { equity += qty; continue; }    // Earn de stable ~ $1
> 152	                const upx = priceOf(under);

> 157	            cold.push(coin); continue;                              // preço frio REAL → fail-safe (não avalia)

> 165	// DECISÃO PURA do cap (sem efeitos colaterais) — testável na suíte de sim. Ordena os backstops:
> 166	//   1) PISO DURO primeiro (ultimate): equity <= dayBase×(1-hardPct) → 'hardHalt' (NUNCA auto-resume).
> 167	//   2) AUTO-RESUME: cap armado + cooldown vencido + dentro do limite + não-hardHalted → 'resume'.
> 168	//   3) SOFT cap: equity <= equity-baseline-rolante×(1-capPct) → 'softTrip' (+canResume).
> 169	//   4) senão 'none'. snap=null → 'none'.
> 170	function evaluateCap(snap, equity, nowMs, cfg) {
> 171	    if (!snap) return { action: 'none' };
> 172	    const hardFloor = snap.dayBase * (1 - cfg.hardPct);
> 173	    if (equity <= hardFloor) return { action: 'hardHalt', hardFloor };
> 174	    if (cfg.autoResume && cfg.capActive && !snap.hardHalted && snap.resumeAt && nowMs >= snap.resumeAt && snap.resumes < cfg.maxResumes)
> 175	        return { action: 'resume' };
> 176	    const softFloor = snap.equity * (1 - cfg.capPct);
> 177	    if (equity <= softFloor) return { action: 'softTrip', softFloor, canResume: !!(cfg.autoResume && snap.resumes < cfg.maxResumes) };
> 178	    return { action: 'none' };
> 179	}
