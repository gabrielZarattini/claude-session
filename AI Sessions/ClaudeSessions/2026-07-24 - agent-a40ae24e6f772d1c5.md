> 230	v_git_commit() {
> 231	  local val="$1" t subj
> 232	  if ! have git; then EVIDENCE='git não está no PATH'; return 2; fi
> 233	  if ! git -C "$ROOT" rev-parse --is-inside-work-tree >/dev/null 2>&1; then
> 234	    EVIDENCE="não é um repositório git: $ROOT"; return 2
> 235	  fi
> 236	  if [ -z "$val" ]; then EVIDENCE='claim sem `value`'; return 1; fi
> 237	  t="$(git -C "$ROOT" cat-file -t "$val" 2>/dev/null || true)"
> 238	  if [ "$t" != "commit" ]; then
> 239	    EVIDENCE="git cat-file -t ${val} → '${t:-<objeto inexistente>}' (esperado: commit)"
> 240	    return 1
> 241	  fi
> 242	  subj="$(git -C "$ROOT" log -1 --format='%h %ad %s' --date=short "$val" 2>/dev/null || true)"
> 243	  EVIDENCE="git cat-file -t → commit · $subj"
> 244	  return 0
> 245	}

> 247	v_file_exists() {
> 248	  local val="$1" abs
> 249	  if [ -z "$val" ]; then EVIDENCE='claim sem `value`'; return 1; fi
> 250	  case "$val" in /*) abs="$val" ;; *) abs="$ROOT/$val" ;; esac
> 251	  if [ ! -e "$abs" ]; then
> 252	    EVIDENCE="não existe no disco: $val"
> 253	    return 1
> 254	  fi
> 255	  EVIDENCE="$(ls -la -- "$abs" 2>/dev/null | sed -e 's/[[:space:]]\{1,\}/ /g' | head -n 1)"
> 256	  return 0
> 257	}

> 259	v_command_exit() {
> 260	  local val="$1" expect="$2" rc=0 log tail_txt
> 261	  if [ "$NO_EXEC" -eq 1 ]; then
> 262	    EVIDENCE='re-execução desabilitada por --no-exec'
> 263	    return 2
> 264	  fi
> 265	  if [ -z "$val" ]; then EVIDENCE='claim sem `value`'; return 1; fi
> 266	  [ -n "$expect" ] || expect=0
> 267	  case "$expect" in ''|*[!0-9-]*) expect=0 ;; esac
> 268	  log="$(mktemp "${TMPDIR:-/tmp}/harness-proof-cmd.XXXXXX")"
> 269	  ( cd "$ROOT" && eval "$val" ) >"$log" 2>&1 || rc=$?
> 270	  tail_txt="$(tail -n 2 "$log" 2>/dev/null | tr '\n' ' ' | cut -c1-160)"
> 271	  rm -f "$log"
> 272	  if [ "$rc" -eq "$expect" ]; then
> 273	    EVIDENCE="re-executado agora → exit $rc (esperado $expect) · $tail_txt"
> 274	    return 0
> 275	  fi
> 276	  EVIDENCE="re-executado agora → exit $rc, esperado $expect · $tail_txt"
> 277	  return 1
> 278	}

> 280	v_http_status() {
> 281	  local val="$1" expect="$2" code=''
> 282	  [ -n "$expect" ] || expect=200
> 283	  case "$expect" in ''|*[!0-9]*) expect=200 ;; esac
> 284	  if [ -z "$val" ]; then EVIDENCE='claim sem `value`'; return 1; fi
> 285	  if have curl; then
> 286	    code="$(curl -s -o /dev/null -w '%{http_code}' --max-time "$HTTP_TIMEOUT" -- "$val" 2>/dev/null || true)"
> 287	  elif have wget; then
> 288	    code="$(wget --server-response --spider -T "$HTTP_TIMEOUT" -t 1 "$val" 2>&1 \
> 289	            | awk '/^[[:space:]]*HTTP\//{ s=$2 } END{ print s }' || true)"
> 290	  else
> 291	    EVIDENCE='nem curl nem wget no PATH'
> 292	    return 2
> 293	  fi
> 294	  if [ -z "$code" ] || [ "$code" = "000" ]; then
> 295	    EVIDENCE="requisição não completou (timeout ${HTTP_TIMEOUT}s ou host inalcançável)"
> 296	    return 1
> 297	  fi
> 298	  if [ "$code" = "$expect" ]; then
> 299	    EVIDENCE="HTTP $code (esperado $expect)"
> 300	    return 0
> 301	  fi
> 302	  EVIDENCE="HTTP $code, esperado $expect"
> 303	  return 1
> 304	}

> 311	  fi

> 321	# ── auditoria de um manifesto ─────────────────────────────────────────────────
> 322	N_OK=0; N_BAD=0; N_NV=0; N_CLAIMS=0
> 323	
> 324	audit_manifest() {
> 325	  local file="$1" tsv line type value expect claim rc
> 326	  printf '\n── %s ──\n' "${file#"$ROOT"/}"
> 327	
> 328	  tsv="$(mktemp "${TMPDIR:-/tmp}/harness-proof-tsv.XXXXXX")"
> 329	  manifest_claims "$file" > "$tsv" 2>/dev/null || true
> 330	
> 331	  if [ ! -s "$tsv" ]; then
> 332	    if grep -qE '"type"[[:space:]]*:' "$file" 2>/dev/null; then
> 333	      rm -f "$tsv"
> 334	      printf 'proof-audit: %s contém claims, mas a leitura devolveu ZERO.\n' "$file" >&2
> 335	      printf '             Auditoria INCONCLUSIVA — instale jq/python3/node.\n' >&2
> 336	      printf '             Não vou reportar verde sobre um manifesto que não consegui ler.\n' >&2
> 337	      exit 3
> 338	    fi
> 339	    rm -f "$tsv"
> 340	    printf '  manifesto sem nenhuma claim.\n'
> 341	    printf '  Um selo sem prova declarada não é um selo verificado — é um selo confiado.\n'
> 342	    N_BAD=$((N_BAD + 1))
> 343	    return 0
> 344	  fi
> 345	
> 346	  while IFS= read -r line; do
> 347	    [ -n "$line" ] || continue
> 348	    type="$(printf '%s' "$line" | cut -f1)"
> 349	    value="$(printf '%s' "$line" | cut -f2)"
> 350	    expect="$(printf '%s' "$line" | cut -f3)"
> 351	    claim="$(printf '%s' "$line" | cut -f4-)"
> 352	    N_CLAIMS=$((N_CLAIMS + 1))
> 353	
> 354	    EVIDENCE=''
> 355	    rc=0
> 356	    case "$type" in
> 357	      git_commit)   v_git_commit   "$value" || rc=$? ;;
> 358	      file_exists)  v_file_exists  "$value" || rc=$? ;;
> 359	      command_exit) v_command_exit "$value" "$expect" || rc=$? ;;
> 360	      http_status)  v_http_status  "$value" "$expect" || rc=$? ;;
> 361	      record_id)    v_record_id    "$value" || rc=$? ;;
> 362	      '')           rc=2; EVIDENCE='claim sem `type`' ;;
> 363	      *)            rc=2; EVIDENCE="tipo desconhecido '$type' — nenhum verificador corresponde" ;;
> 364	    esac
> 365	
> 366	    case "$rc" in
> 367	      0)
> 368	        N_OK=$((N_OK + 1))
> 369	        printf '  [ SUSTENTADA   ] %-13s %s\n' "$type" "${claim:-<sem afirmação>}"
> 370	        printf '                    prova: %s\n' "$EVIDENCE"
> 371	        ;;
> 372	      1)
> 373	        N_BAD=$((N_BAD + 1))
> 374	        printf '  [ NÃO SE SUSTENTA ] %-11s %s\n' "$type" "${claim:-<sem afirmação>}"
> 375	        printf '                    valor: %s\n' "${value:-<vazio>}"
> 376	        printf '                    achado: %s\n' "$EVIDENCE"
> 377	        ;;
> 378	      *)
> 379	        N_NV=$((N_NV + 1))
> 380	        printf '  [ NÃO VERIFICÁVEL ] %-11s %s\n' "$type" "${claim:-<sem afirmação>}"
> 381	        printf '                    valor: %s\n' "${value:-<vazio>}"
> 382	        printf '                    motivo: %s\n' "$EVIDENCE"
> 383	        if [ "$STRICT" -eq 1 ]; then
> 384	          printf '                    → CONTA COMO FALHA (%s)\n' "$STRICT_WHY"
> 385	        else
> 386	          printf '                    → apenas aviso (%s)\n' "$STRICT_WHY"
> 387	        fi
> 388	        ;;
> 389	    esac
> 390	  done < "$tsv"
> 391	  rm -f "$tsv"
> 392	  return 0
> 393	}
