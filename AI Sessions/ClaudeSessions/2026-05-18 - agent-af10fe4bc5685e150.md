> 587	 * Translate a typed node's data into the request body accepted by the
> 588	 * `canvas-execute` Edge Function. Soul models split into provider/model
> 589	 * plus parameters.resolution/batch so the function's tiering helper
> 590	 * (`resolveHiggsfieldSoulKey`) computes the correct cost.
> 591	 */
> 592	export function resolveExecutePayload(
> 593	  data: CanvasNodeData,
> 594	): {
> 595	  node_type: string;
> 596	  provider: ProviderId;
> 597	  model: string;
> 598	  prompt: string;
> 599	  negative_prompt?: string;
> 600	  parameters: Record<string, unknown>;
> 601	} | null {
> 602	  switch (data.kind) {
> 603	    case "generateImage": {
> 604	      const info = MODEL_INFO[data.modelId];
> 605	      const parameters: Record<string, unknown> = {
> 606	        aspect_ratio: data.aspectRatio,
> 607	        controls: data.controls,
> 608	        seed: data.fixedSeed && data.seed != null ? data.seed : undefined,
> 609	      };
> 610	      if (info.resolution) parameters.resolution = info.resolution;
> 611	      if (info.batch) parameters.batch = info.batch;
> 612	      return {
> 613	        node_type: "generate_image",
> 614	        provider: info.provider,
> 615	        model: info.modelParam,
> 616	        prompt: data.prompt,
> 617	        negative_prompt: data.negativePrompt || undefined,
> 618	        parameters,
> 619	      };
> 620	    }
> 621	    case "sceneCompose": {
> 622	      const info = MODEL_INFO[data.modelId];
> 623	      const parameters: Record<string, unknown> = {
> 624	        aspect_ratio: data.aspectRatio,
> 625	      };
> 626	      if (info.resolution) parameters.resolution = info.resolution;
> 627	      if (info.batch) parameters.batch = info.batch;
> 628	      return {
> 629	        node_type: "scene_compose",
> 630	        provider: info.provider,
> 631	        model: info.modelParam,
> 632	        prompt: data.prompt,
> 633	        parameters,
> 634	      };
> 635	    }
> 636	    case "styleTransfer": {
> 637	      const preset = STYLE_TRANSFER_PRESETS.find((p) => p.id === data.presetId);
> 638	      return {
> 639	        node_type: "style_transfer",
> 640	        provider: "higgsfield",
> 641	        model: "soul",
> 642	        prompt: preset?.suffix ?? "",
> 643	        parameters: {
> 644	          strength: data.strength,
> 645	          preset_id: data.presetId,
> 646	          resolution: "720p",
> 647	          batch: 1,
> 648	        },
> 649	      };
> 650	    }
> 651	    case "imageToVideo": {
> 652	      const info = VIDEO_MODEL_INFO[data.videoModel];
> 653	      return {
> 654	        node_type: "image_to_video",
> 655	        provider: info.provider,
> 656	        model: info.modelParam,
