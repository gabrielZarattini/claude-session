```javascript
async function updateMemory(symbol, index, interval, value, executeAutomations = true) {
    // symbol: string 'BTCUSDT'
    // index: string (indexKey enum, ex. 'REGIME', 'FLOW', 'BOOK')
    // interval: null ou string '1h', '1m', etc.
    // value: qualquer — número, string, objeto; será serializado em MEMORY
    // executeAutomations: BOOLEANO (default true)
    //   = true  → disparar automações após escrever (testAutomations(memoryKey))
    //   = false → escrever PASSIVAMENTE, SEM disparar (gate lido no tick, não gatilho)
    
    if (value === undefined || value === null) return false;
    if (value.toJSON) value = value.toJSON();
    if (value.get) value = value.get({ plain: true });
    
    if (LOCK_MEMORY) return false;  // trava global
    
    const memoryKey = parseMemoryKey(symbol, index, interval);  // 'SYMBOL:INDEX' ou 'SYMBOL:INDEX_INTERVAL'
    MEMORY[memoryKey] = value;  // escreve em MEMORY
    
    if (LOGS) logger('beholder', `Beholder memory updated: ${memoryKey} => ${JSON.stringify(value)}, will exec autos? ${executeAutomations}`);
    
    if (!executeAutomations) return false;  // *** SALTA AUTOMAÇÕES ***
    
    return testAutomations(memoryKey);  // retorna resultado das automações que dispararam
}
```

#### Assinatura de getMemory (linhas 893-903)
```javascript
function getMemory(symbol, index, interval) {
    if (symbol && index) {
        const indexKey = interval ? `${index}_${interval}` : index;
        const memoryKey = `${symbol}:${indexKey}`;
        const result = MEMORY[memoryKey];
        return typeof result === 'object' ? { ...result } : result;  // retorna cópia se objeto
    }
    return { ...MEMORY };  // sem args: retorna TODA a MEMORY
}
```

```javascript
    return `${symbol}:${indexKey}`;  // 'BTCUSDT:REGIME' ou 'BTCUSDT:FLOW_1m'
}
```
