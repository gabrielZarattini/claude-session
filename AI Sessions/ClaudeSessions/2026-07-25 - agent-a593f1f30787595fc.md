> 664	function parseMarkdownAndJson(text: string) {
> 665	  const trimmed = text.trim();
> 666	  // Try to parse global JSON first
> 667	  if ((trimmed.startsWith("{") && trimmed.endsWith("}")) || (trimmed.startsWith("[") && trimmed.endsWith("]"))) {
> 668	    try {
> 669	      const parsed = JSON.parse(trimmed);
> 670	      return { type: "json" as const, data: parsed };
> 671	    } catch {
> 672	      // Not a valid JSON, fallback to markdown
> 673	    }
> 674	  }
> 675	
> 676	  // Robust block splitter that supports streaming codeblocks
> 677	  const blocks: Array<{ type: "code" | "markdown"; content: string; lang?: string }> = [];
> 678	  let i = 0;
> 679	  while (i < text.length) {
> 680	    const nextCodeBlockStart = text.indexOf("```", i);
> 681	    if (nextCodeBlockStart === -1) {
> 682	      const content = text.slice(i);
> 683	      if (content) {
> 684	        blocks.push({ type: "markdown", content });
> 685	      }
> 686	      break;
> 687	    }
> 688	
> 689	    if (nextCodeBlockStart > i) {
> 690	      blocks.push({ type: "markdown", content: text.slice(i, nextCodeBlockStart) });
> 691	    }
> 692	
> 693	    const startOfCode = nextCodeBlockStart + 3;
> 694	    const endOfCode = text.indexOf("```", startOfCode);
> 695	    const newlineIndex = text.indexOf("\n", startOfCode);
> 696	    let lang = "";
> 697	    let codeStart = startOfCode;
> 698	
> 699	    if (newlineIndex !== -1 && (endOfCode === -1 || newlineIndex < endOfCode)) {
> 700	      lang = text.slice(startOfCode, newlineIndex).trim();
> 701	      codeStart = newlineIndex + 1;
> 702	    }
> 703	
> 704	    if (endOfCode === -1) {
> 705	      // Unclosed code block (streaming)
> 706	      const content = text.slice(codeStart);
> 707	      blocks.push({ type: "code", content, lang });
> 708	      break;
> 709	    } else {
> 710	      const content = text.slice(codeStart, endOfCode);
> 711	      blocks.push({ type: "code", content, lang });
> 712	      i = endOfCode + 3;
> 713	    }
> 714	  }
> 715	
> 716	  return { type: "markdown" as const, blocks };
> 717	}
