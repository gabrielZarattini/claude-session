> 132	
> 133	Sem Dono declarado não há accountability material → falha vira custo invisível.

> 194	# Ecoa "<slug>\t<regra>" para que --explain possa mostrar QUAL regra derivou o slug.
> 195	derive_slug() {
> 196	  local path="$1" pat="$2" prefix rel first rule
> 197	  prefix="$(static_prefix "$pat")"
> 198	  if [ -n "$prefix" ] && [ "${path#"$prefix"}" != "$path" ]; then
> 199	    rel="${path#"$prefix"}"
> 200	    rule="prefixo-literal:${prefix}"

> 202	    rel="$(basename -- "$(dirname -- "$path")")"
> 203	    [ "$rel" = "." ] && rel="$(basename -- "$path")"

> 205	  fi
> 206	  first="${rel%%/*}"
> 207	  first="${first%.*}"
> 208	  printf '%s\t%s' "$(slugify "$first")" "$rule"

> 246	# Retorna 0/1/2 conforme o contrato. Imprime o veredito e, quando bloqueia,
> 247	# reproduz o Modo de Recusa de 01-closed-loop.md §6 em stderr.
> 248	judge_module() {
> 249	  local slug="$1" suite status base doc missing=() present=0
> 250	  suite="$(suite_dir_for "$slug")"
> 251	
> 252	  if [ ! -d "$suite" ]; then
> 253	    _fail "módulo '$slug' — suíte AUSENTE em $(rel "$suite")"
> 254	    refusal "$slug" "$suite" 2 "a suíte não existe"
> 255	    return 2
> 256	  fi
> 257	
> 258	  for base in "${REQUIRED[@]}"; do
> 259	    if doc="$(find_doc "$suite" "$base")"; then
> 260	      present=$((present + 1))
> 261	    else
> 262	      missing+=("$base")
> 263	    fi
> 264	  done
> 265	
> 266	  if [ "${#missing[@]}" -gt 0 ]; then
> 267	    _fail "módulo '$slug' — suíte INCOMPLETA: ${present}/${#REQUIRED[@]} documentos requeridos"
> 268	    printf '  ausentes: %s\n' "${missing[*]}" >&2
> 269	    refusal "$slug" "$suite" 2 "faltam documentos requeridos (${missing[*]})"
> 270	    return 2
> 271	  fi
> 272	
> 273	  local index
> 274	  if ! index="$(find_doc "$suite" "${REQUIRED[0]}")"; then
> 275	    _fail "módulo '$slug' — índice '${REQUIRED[0]}' não encontrado em $(rel "$suite")"
> 276	    refusal "$slug" "$suite" 2 "índice ausente"
> 277	    return 2
> 278	  fi
> 279	
> 280	  status="$(read_gate_status "$index")"
> 281	  case " $STATES_AUTHORIZE " in
> 282	    *" $status "*)
> 283	      _ok "módulo '$slug' — GATE: $status → implementação AUTORIZADA (${present}/${#REQUIRED[@]} docs · $(rel "$index"))"
> 284	      printf '     nota: a existência dos %d documentos é condição NECESSÁRIA, não suficiente;\n' "${#REQUIRED[@]}"
> 285	      printf '           quem autoriza é o marcador GATE, e ele foi lido de %s\n' "$(rel "$index")"
> 286	      return 0
> 287	      ;;
> 288	  esac
> 289	
> 290	  case "$status" in
> 291	    __missing__)
> 292	      _fail "módulo '$slug' — marcador GATE AUSENTE nas primeiras $MARKER_SCAN_LINES linhas de $(rel "$index")"
> 293	      printf '  ausência de marcador NUNCA é tratada como `open`: sem estado legível, a suíte é\n' >&2
> 294	      printf '  rascunho por definição. Acrescente ao topo do índice: <!-- GATE: draft -->\n' >&2
> 295	      refusal "$slug" "$suite" 1 "marcador ausente (tratado como draft)"
> 296	      return 1
> 297	      ;;
> 298	    __unknown__)
> 299	      _fail "módulo '$slug' — marcador GATE presente porém irreconhecível em $(rel "$index")"
> 300	      printf '  estados válidos: %s %s\n' "$STATES_BLOCK" "$STATES_AUTHORIZE" >&2
> 301	      refusal "$slug" "$suite" 1 "marcador irreconhecível"
> 302	      return 1
> 303	      ;;
> 304	    *)
> 305	      _fail "módulo '$slug' — GATE: $status → implementação BLOQUEADA"
> 306	      refusal "$slug" "$suite" 1 "GATE: $status"
> 307	      return 1
> 308	      ;;
> 309	  esac
> 310	}

> 344	resolve_touched_modules() {
> 345	  local path pat slug rule pair seen=""
> 346	  while IFS= read -r path; do
