> 159	/** Stream SSE from edge function using user's auth token */
> 160	async function streamFromEdge(
> 161	  fnName: string,
> 162	  body: object,
> 163	  onChunk: (text: string) => void,
> 164	): Promise<string> {
> 165	  const token = await getAuthToken();
> 166	  const resp = await fetch(
> 167	    `${import.meta.env.VITE_SUPABASE_URL}/functions/v1/${fnName}`,
> 168	    {
> 169	      method: "POST",
> 170	      headers: {
> 171	        "Content-Type": "application/json",
> 172	        Authorization: `Bearer ${token}`,
> 173	      },
> 174	      body: JSON.stringify(body),
> 175	    }
> 176	  );
> 177	
> 178	  if (!resp.ok) {
> 179	    const err = await resp.json();
> 180	    throw new Error(err.error || "Erro na geração");
> 181	  }
> 182	
> 183	  const reader = resp.body?.getReader();
> 184	  if (!reader) throw new Error("No stream");
> 185	  const decoder = new TextDecoder();
> 186	  let buffer = "";
> 187	  let result = "";
> 188	
> 189	  while (true) {
> 190	    const { done, value } = await reader.read();
> 191	    if (done) break;
> 192	    buffer += decoder.decode(value, { stream: true });
> 193	
> 194	    let nl: number;
> 195	    while ((nl = buffer.indexOf("\n")) !== -1) {
> 196	      let line = buffer.slice(0, nl);
> 197	      buffer = buffer.slice(nl + 1);
> 198	      if (line.endsWith("\r")) line = line.slice(0, -1);
> 199	      if (!line.startsWith("data: ")) continue;
> 200	      const jsonStr = line.slice(6).trim();
> 201	      if (jsonStr === "[DONE]") break;
> 202	      try {
> 203	        const parsed = JSON.parse(jsonStr);
> 204	        const content = parsed.choices?.[0]?.delta?.content;
> 205	        if (content) {
> 206	          result += content;
> 207	          onChunk(result);
> 208	        }
> 209	      } catch {}
> 210	    }
> 211	  }
> 212	  return result;
> 213	}

### 3. **Creative Metrics Schema: Enrich vs. New Table**

**Current schema:**
- `creative_metrics` is keyed by `(content_variant_id, source)` where source ∈ {'organic', 'affiliate_click', 'postback'}
- Columns: `impressions`, `engagements`, `affiliate_clicks`, `revenue_cents`, `collected_at`, `platform`
- **One row per creative per source type** — aggregated snapshot, no time-series yet

**Analysis:**

For a **per-publication analytics dashboard** (views/likes/comments/shares/reposts/saves), you face a tradeoff:

**(A) Enrich `creative_metrics` with platform-specific fields:**
- Add columns: `likes INT`, `comments INT`, `shares INT`, `reposts INT`, `saves INT`, `external_post_id TEXT`
- PRO: Single table, simpler joins
- CON: Violates normalization (organic, affiliate_click, postback sources have different semantics for these fields); unclear which source row to use for UI display
- CON: No time-series — you only have one snapshot per creative per source, not engagement over time

**(B) Create dedicated `publication_metrics` table (RECOMMENDED for time-series):**

- Metrics evolve over days/weeks (a post gains likes for a month)
- `snapshot_at` enables charting engagement growth over time (hour-by-hour, day-by-day)
- Avoids polluting the affiliate-focused `creative_metrics` schema
- Follows the principle of separate concerns: `creative_metrics` = revenue loop; `publication_metrics` = social performance

**Collector strategy:** Write one row per content per platform per collection window (e.g., daily at 6 AM UTC + on-demand polls). Index on `(content_variant_id, platform, snapshot_at DESC)` for fast time-series queries.
