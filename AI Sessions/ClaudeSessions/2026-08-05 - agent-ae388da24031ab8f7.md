> 300	/**
> 301	 * Montagem do MASTER — concatena os clipes na ordem recebida e normaliza o loudness.
> 302	 *
> 303	 * Por que re-encodar em vez de usar concat demuxer com `-c copy`: as peças vêm de motores
> 304	 * diferentes (motion 1280x720 sem áudio, Veo com áudio, cortes com outro fps). Stream copy só
> 305	 * funciona com parâmetros idênticos — aqui ele produziria um arquivo quebrado ou mudo. O filtro
> 306	 * `concat` com normalização de escala/fps/áudio é o caminho honesto.
> 307	 *
> 308	 * Cena sem áudio ganha silêncio (anullsrc): sem isso o `concat` com `a=1` falha ou desalinha —
> 309	 * é a mesma armadilha que truncou a faixa do EP05 ("não confiar no áudio do concat").
> 310	 */
> 311	async function processAssemble(renderId: string, userId: string, comp: Record<string, unknown>) {
> 312	  const workDir = join(WORK_BASE, renderId);
> 313	  rmSync(workDir, { recursive: true, force: true });
> 314	  mkdirSync(workDir, { recursive: true });
> 315	  try {
> 316	    const clips = (comp.clips ?? []) as Array<{ bucket: string; key: string }>;
> 317	    if (!clips.length) throw new Error("sem clipes");
> 318	    const aspect = comp.aspect === "9:16" ? "9:16" : "16:9";
> 319	    const W = aspect === "9:16" ? 1080 : 1920;
> 320	    const H = aspect === "9:16" ? 1920 : 1080;
> 321	    const lufs = Number(comp.loudnorm_lufs) || -16;
> 322	
> 323	    // baixa cada clipe (a ordem da lista É a ordem do filme)
> 324	    const locals: string[] = [];
> 325	    for (let i = 0; i < clips.length; i++) {
> 326	      const { data, error } = await supabase.storage.from(clips[i].bucket).download(clips[i].key);
> 327	      if (error || !data) { console.error(`⚠️  clipe ${i} indisponível: ${error?.message}`); continue; }
> 328	      const local = join(workDir, `c${String(i).padStart(3, "0")}.mp4`);
> 329	      writeFileSync(local, Buffer.from(await data.arrayBuffer()));
> 330	      locals.push(local);
> 331	    }
> 332	    if (!locals.length) throw new Error("nenhum clipe baixou");
> 333	
> 334	    // Normaliza cada entrada e concatena VÍDEO E ÁUDIO (assemble-graph.ts). O graph anterior
> 335	    // criava streams de silêncio [sN] nunca consumidos (concat a=0 + map só do vídeo) —
> 336	    // filter_complex com output órfão é ERRO FATAL no ffmpeg, e mesmo sem o erro o master sairia

> 344	      "ffmpeg", "-y", ...inputs,
> 345	      "-filter_complex", filter,
> 346	      "-map", maps[0], "-map", maps[1],
> 347	      "-c:v", "libx264", "-pix_fmt", "yuv420p", "-crf", "19", "-preset", "medium",
> 348	      "-c:a", "aac", "-b:a", "192k",
> 349	      "-movflags", "+faststart",
> 350	      out, "-loglevel", "error",
> 351	    ], 40 * 60 * 1000);
> 352	
> 353	    const bytes = readFileSync(out);
> 354	    if (bytes.length < 50_000) throw new Error(`master pequeno demais (${bytes.length}B)`);
> 355	    const probe = Bun.spawnSync(["ffprobe", "-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", out]);
> 356	    const dur = Number(new TextDecoder().decode(probe.stdout).trim()) || 0;

> 361	    if (upErr) throw new Error(`upload: ${upErr.message}`);
> 362	
> 363	    const reg = await supabase.rpc("register_creative_asset", {
> 364	      p_user_id: userId, p_kind: "video",
> 365	      p_storage_bucket: BUCKET, p_storage_key: key,

> 368	      p_metadata: {
> 369	        project_id: comp.project_id ?? null, node_id: comp.node_id ?? null,
> 370	        master: { scenes: locals.length, duration_sec: Math.round(dur), aspect, loudnorm_lufs: lufs },

> 388	/** Devolve à fila jobs 'running' presos além do teto (worker morto no meio). */
