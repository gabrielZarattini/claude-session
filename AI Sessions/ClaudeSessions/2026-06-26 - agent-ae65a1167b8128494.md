**View Definition** (`decrypted_social_accounts`, line 564-581 of encryption migration):
```sql
CREATE OR REPLACE VIEW public.decrypted_social_accounts AS
SELECT 
  sa.id, sa.user_id, sa.platform, sa.platform_user_id, sa.platform_username,
  COALESCE(s_access.decrypted_secret, sa.access_token) AS access_token,  -- Vault LEFT JOIN
  COALESCE(s_refresh.decrypted_secret, sa.refresh_token) AS refresh_token,
  sa.token_expires_at, sa.scopes, sa.is_active, sa.metadata, ...
FROM public.social_accounts_table sa
LEFT JOIN vault.decrypted_secrets s_access ON sa.access_token ~ '^[0-9a-f-]{36}$' AND s_access.id = sa.access_token::uuid
LEFT JOIN vault.decrypted_secrets s_refresh ON sa.refresh_token ~ '^[0-9a-f-]{36}$' AND s_refresh.id = sa.refresh_token::uuid;
```

```sql
COALESCE(s_token.decrypted_secret, mc.long_lived_token) AS long_lived_token,
COALESCE(s_pages.decrypted_secret::jsonb, mc.pages) AS pages,
```
