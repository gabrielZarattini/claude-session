> 216	function getInsideCandle(ohlc, last, bars) {
> 217	    let hasInsideCandle = ohlc.high[last] < ohlc.high[last - 1] && ohlc.low[last] > ohlc.low[last - 1];
> 218	    if (hasInsideCandle && bars > 1) {
> 219	        for (let i = 1; i < bars; i++)
> 220	            hasInsideCandle = hasInsideCandle && ohlc.high[last - i] < ohlc.high[last - i - 1] && ohlc.low[last - i] > ohlc.low[last - i - 1];
> 221	    }
> 222	    return hasInsideCandle;
> 223	}
