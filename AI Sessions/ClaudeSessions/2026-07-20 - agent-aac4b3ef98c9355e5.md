> 120	  file: File,
> 121	  onProgress: (pct: number) => void,
> 122	): Promise<{ filename: string; bytes: number }> {
> 123	  const { data: { session } } = await supabase.auth.getSession();
> 124	  const token = session?.access_token;
> 125	  if (!token) throw new Error('Sessão expirada — faça login novamente.');
> 126	  // Headers must be ASCII — encode spaces/accents in the filename (server decodeURIComponent + sanitize).
> 127	  // X-Total-Bytes deixa o servidor VERIFICAR a integridade do arquivo montado antes de aceitá-lo.
> 128	  // Sem isso, um chunk perdido/duplicado produz um arquivo com tamanho plausível e mídia corrompida

> 130	  const base: Record<string, string> = {
> 131	    Authorization: `Bearer ${token}`,
> 132	    'X-Filename': encodeURIComponent(file.name),
> 133	    'X-Total-Bytes': String(file.size),
> 134	  };
> 135	  const total = file.size;
> 136	  onProgress(0);
> 137	
> 138	  if (total <= CHUNK_BYTES) {
> 139	    const r = await xhrPost(file, base, (loaded) => onProgress(Math.round((loaded / total) * 100)));
> 140	    return { filename: r.filename ?? file.name, bytes: r.bytes ?? total };
> 141	  }
> 142	
> 143	  const uploadId = crypto.randomUUID ? crypto.randomUUID() : `${Date.now()}-${Math.round(Math.random() * 1e9)}`;
> 144	  const count = Math.ceil(total / CHUNK_BYTES);
> 145	  let uploaded = 0;
> 146	  let result: UploadResp = {};
> 147	  for (let i = 0; i < count; i++) {
> 148	    const blob = file.slice(i * CHUNK_BYTES, Math.min((i + 1) * CHUNK_BYTES, total));
> 149	    result = await xhrPost(
> 150	      blob,
> 151	      { ...base, 'X-Upload-Id': uploadId, 'X-Chunk-Index': String(i), 'X-Chunk-Count': String(count) },
> 152	      (loaded) => onProgress(Math.round(((uploaded + loaded) / total) * 100)),
> 153	    );
> 154	    uploaded += blob.size;
> 155	    onProgress(Math.round((uploaded / total) * 100));
> 156	  }
> 157	  return { filename: result.filename ?? file.name, bytes: result.bytes ?? total };
