> 221	function getInsideCandle(ohlc, last, bars) {
> 222	    let hasInsideCandle = ohlc.high[last] < ohlc.high[last - 1] && ohlc.low[last] > ohlc.low[last - 1];
> 223	    if (hasInsideCandle && bars > 1) {
> 224	        for (let i = 1; i < bars; i++)
> 225	            hasInsideCandle = hasInsideCandle && ohlc.high[last - i] < ohlc.high[last - i - 1] && ohlc.low[last - i] > ohlc.low[last - i - 1];
> 226	    }
> 227	    return hasInsideCandle;
> 228	}
