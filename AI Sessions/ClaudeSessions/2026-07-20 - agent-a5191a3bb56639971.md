> 152	  images: string[];            // ordered absolute paths to pre-generated frames (PNG/JPG)

> 156	  height?: number;             // canvas height (default 1920 → 9:16)

> 159	export async function encodeImageSequence(spec: ImageSequenceSpec, outPath: string): Promise<RenderResult> {
> 160	  const started = Date.now();
> 161	  const images = (spec.images ?? []).map((p) => resolve(p));
> 162	  if (images.length === 0) throw new Error('encodeImageSequence: no images provided');
> 163	  for (const p of images) if (!existsSync(p)) throw new Error(`encodeImageSequence: missing frame ${p}`);
> 164	  const fps = spec.fps ?? 30;
> 165	  const secPer = spec.seconds_per_image ?? 2.5;
> 166	  const W = spec.width ?? 1080;
> 167	  const H = spec.height ?? 1920;
> 168	
> 169	  // FFmpeg concat demuxer: hold each image for `secPer`s. The demuxer drops the final entry's duration,
> 170	  // so the last file is repeated once (documented ffmpeg quirk) to honor its hold time.
> 171	  const listDir = mkdtempSync(join(tmpdir(), 'hf-seq-'));
> 172	  const esc = (p: string) => p.replace(/'/g, "'\\''");
> 173	  const lines: string[] = [];
> 174	  for (const p of images) { lines.push(`file '${esc(p)}'`); lines.push(`duration ${secPer}`); }
> 175	  lines.push(`file '${esc(images[images.length - 1])}'`);
> 176	  const listPath = join(listDir, 'frames.txt');
> 177	  writeFileSync(listPath, lines.join('\n') + '\n');
> 178	
> 179	  try {
> 180	    await runFfmpeg([
> 181	      '-y',
> 182	      '-f', 'concat', '-safe', '0', '-i', listPath,
> 183	      // scale to fit, pad to the exact 9:16 canvas (letterbox, no crop), normalize SAR, fix fps
> 184	      '-vf', `scale=${W}:${H}:force_original_aspect_ratio=decrease,pad=${W}:${H}:(ow-iw)/2:(oh-ih)/2:color=black,setsar=1,fps=${fps}`,
> 185	      '-c:v', 'libx264', '-preset', 'medium', '-crf', '20',
> 186	      '-pix_fmt', 'yuv420p', '-r', String(fps),
> 187	      '-movflags', '+faststart', '-fflags', '+bitexact', '-flags:v', '+bitexact', '-map_metadata', '-1',
> 188	      outPath,
> 189	    ]);
> 190	  } finally {
> 191	    rmSync(listDir, { recursive: true, force: true });
> 192	  }
> 193	  if (!existsSync(outPath) || statSync(outPath).size < 1024) throw new Error('encodeImageSequence: output missing or too small');
> 194	  const probed = await ffprobeVideo(outPath);
> 195	  return {
> 196	    outPath,
> 197	    bytes: statSync(outPath).size,
> 198	    width: probed.width,
> 199	    height: probed.height,
> 200	    frames: probed.frames,
> 201	    fps: probed.fps,
> 202	    durationMs: Date.now() - started,
> 203	  };
