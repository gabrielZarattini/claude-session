```ts
// resolveMaster: quando existsSync(path) for false, tentar Storage antes de desistir.
async function resolveMaster(userId, sourceKey) {
  // ...guards existentes :98-100...
  if (existsSync(path)) return { path, size: statSync(path).size, cleanup: () => {} };

  // Fallback Storage: resolver bucket/key pela linha owner-scoped.
  const { data: asset } = await supabase.from('creative_assets')
    .select('storage_bucket, storage_key')
    .eq('storage_key', sourceKey).eq('user_id', userId).maybeSingle();
  if (!asset || asset.storage_bucket === 'local') return { error: 'source_not_found' };

  const { data: blob, error } = await supabase.storage
    .from(asset.storage_bucket).download(asset.storage_key);
  if (error || !blob) return { error: 'source_download_failed' };
  const tmp = join(tmpdir(), `yt-${randomUUID()}.mp4`);
  writeFileSync(tmp, Buffer.from(await blob.arrayBuffer()));  // stream-to-disk; mantém o piso resumable de 8 MB
  return { path: tmp, size: statSync(tmp).size, cleanup: () => rmSync(tmp, { force: true }) };
}
// processJob: chamar master.cleanup() num finally.
```
