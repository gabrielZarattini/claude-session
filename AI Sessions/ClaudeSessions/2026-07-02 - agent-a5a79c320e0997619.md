> 59	// ── field_map grammar (§3): evaluate 'atoms.<token>' expressions joined by '+'. ──────────────────────────────
> 60	function evalExpr(expr: string, atoms: Atoms, virtuals: Record<string, string>, sep: string): string {
> 61	  return String(expr || "")
> 62	    .split("+")
> 63	    .map((s) => s.trim())
> 64	    .map((part) => {
> 65	      if (!part) return "";
> 66	      if ((part.startsWith("'") && part.endsWith("'")) || (part.startsWith('"') && part.endsWith('"'))) {
> 67	        return part.slice(1, -1);
> 68	      }
> 69	      if (part.startsWith("atoms.")) {
> 70	        const name = part.slice("atoms.".length).trim();
> 71	        if (name in virtuals) return virtuals[name];
> 72	        const v = (atoms as Record<string, unknown>)[name];
> 73	        if (Array.isArray(v)) return v.map((x) => String(x)).join("\n");
> 74	        return v == null ? "" : String(v);
> 75	      }
> 76	      return ""; // nested sub-directive (e.g. 'reframe_1.91x1') or unknown token → ignored in TEXT assembly
> 77	    })
> 78	    .filter(Boolean)
> 79	    .join(sep);
> 80	}

> 132	// FR-CP-005/006 — in-process image REFRAME + format coercion (imagescript WASM; no host worker, no FFmpeg).
> 133	// Center-crops the master to the target aspect (safe-zone bias = center; saliency via Vision-MCP is a refinement),
> 134	// resizes to the channel's exact px, and encodes JPEG (the universal API-safe format: IG JPEG-only, TikTok/
> 135	// Pinterest/LinkedIn/FB all accept JPEG). Throws → caller fails open to 'pending_reframe'.
> 136	async function reframeToJpeg(src: Uint8Array, tw: number, th: number): Promise<Uint8Array> {
> 137	  const img = await Image.decode(src);
> 138	  const ta = tw / th;
> 139	  const sa = img.width / img.height;
> 140	  let cw = img.width;
> 141	  let ch = img.height;
> 142	  if (sa > ta) cw = Math.max(1, Math.round(img.height * ta)); // too wide → trim width
> 143	  else ch = Math.max(1, Math.round(img.width / ta));          // too tall → trim height
> 144	  const x = Math.max(0, Math.round((img.width - cw) / 2));
> 145	  const y = Math.max(0, Math.round((img.height - ch) / 2));
> 146	  const out = img.crop(x, y, cw, ch).resize(tw, th);
> 147	  return await out.encodeJPEG(85);
> 148	}
